## Задача 1. Напишите программу, которая проверяет является ли введенный год високосным.
## Код
```c
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <locale.h>

int remainder_by_4(int year);
int is_leap_year(int year);

int main(void)
{
    int year;
    int rem;
    int leap;

    setlocale(LC_ALL, "Russian");

    printf("Введите год: ");
    if (scanf("%d", &year) != 1 || year <= 0)
    {
        printf("Ошибка: нужно ввести целое положительное число.\n");
        return 1;
    }

    /* 1 */
    rem = remainder_by_4(year);
    printf("Остаток от деления на 4: %d\n", rem);
    if (rem == 0)
        printf("По упрощённому правилу (кратность 4): год %d високосный\n", year);
    else
        printf("По упрощённому правилу (кратность 4): год %d не високосный\n", year);

    /* 2 */
    leap = is_leap_year(year);
    if (leap)
        printf("По точному правилу (4, 100, 400): год %d високосный\n", year);
    else
        printf("По точному правилу (4, 100, 400): год %d не високосный\n", year);

    return 0;
}

int remainder_by_4(int year)
{
    int rem;
    rem = year % 4;
    return rem;
}

int is_leap_year(int year)
{
    int leap;
    leap = (year % 4 == 0 && year % 100 != 0) || (year % 400 == 0);
    return leap;
}
```

```

Остаток от деления на 4: 2
По упрощённому правилу (кратность 4): год 2018 не високосный
По точному правилу (4, 100, 400): год 2018 не високосный

Введите год: 1900
Остаток от деления на 4: 0
По упрощённому правилу (кратность 4): год 1900 високосный
По точному правилу (4, 100, 400): год 1900 не високосный

Введите год: 2020

```
## Задача 2. Напишите программу для вычислений значения функции F(x).
## Код
```c
/*
 *          | x^2 - 3x + 9,   при x <= 3
 *   F(x) = |
 *          | 1 / (x^3 + 6),  при x > 3
 *
 * Контрольные примеры (рассчитаны вручную):
 * Ветвь x <= 3:
 * x = -2 -> 4 + 6 + 9  = 19
 * x = 0  -> 0 - 0 + 9  = 9
 * x = 3  -> 9 - 9 + 9  = 9         
 * (граница, входит в первую ветвь)
 * 
 * Ветвь x > 3:
 * x = 3.01 -> 1 / (27.270901 + 6)  = 0.030056   (сразу за границей)
 * x = 4    -> 1 / (64 + 6) = 1/70  = 0.014286
 * x = 5    -> 1 / (125 + 6)= 1/131 = 0.007634
 *
 * Знаменатель x^3 + 6 обращается в 0 только при x ≈ -1.817,
 * а вторая ветвь работает при x > 3, поэтому деления на ноль нет.
 */
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <locale.h>

double f(double x);

int main(void)
{
    double x;
    double result;

    setlocale(LC_ALL, "Russian");

    printf("Введите x: ");
    if (scanf("%lf", &x) != 1)
    {
        printf("Ошибка: нужно ввести число.\n");
        return 1;
    }

    result = f(x);

    /*с использованием тернарной операции в printf */
    printf("F(%g) = %.6f, использована формула %s\n",
        x, result,
        (x <= 3) ? "x^2 - 3x + 9 (x <= 3)" : "1/(x^3 + 6) (x > 3)");

    return 0;
}

double f(double x)
{
    double y;
    if (x <= 3)
        y = x * x - 3 * x + 9;
    else
        y = 1.0 / (x * x * x + 6);
    return y;
}
```
## Результат
```

F(-2) = 19,000000, использована формула x^2 - 3x + 9 (x <= 3)

Введите x: 3
F(3) = 9,000000, использована формула x^2 - 3x + 9 (x <= 3)

Введите x: 3,01
F(3,01) = 0,030056, использована формула 1/(x^3 + 6) (x > 3)

Введите x: 4
F(4) = 0,014286, использована формула 1/(x^3 + 6) (x > 3)
```
